import os
import sys

print("Process Management Program")

pid = os.fork()

if pid == 0:
    print("\nChild Process")
    print("Child Process ID:", os.getpid())
    print("Parent Process ID:", os.getppid())

    print("Child process is executing")
    sys.exit(0)

else:
    print("\nParent Process")
    print("Parent Process ID:", os.getpid())
    print("Child Process ID:", pid)

    os.wait()

    print("Child process completed")
    print("Parent process completed")

wt = [0] * n

for i in range(1, n):

//****SJF****//
. SJF CPU Scheduling Algorithm
Theory
SJF stands for Shortest Job First. It is a non-preemptive CPU scheduling algorithm.
- The process having the smallest burst time gets the CPU first.
- The shortest process is executed before longer processes.
- It generally provides a lower average waiting time than FCFS.
- Once a process starts execution, it continues until completion

n = int(input("Enter number of process: "))

bt = []

for i in range(n):
    b = int(input("Enter burst time: "))
    bt.append(b)

p = []

for i in range(n):
    p.append(i + 1)

for i in range(n):
    for j in range(i + 1, n):
        if bt[i] > bt[j]:
            temp = bt[i]
            bt[i] = bt[j]
            bt[j] = temp

            temp = p[i]
            p[i] = p[j]
            p[j] = temp

wt = [0] * n
tat = [0] * n

for i in range(1, n):
    wt[i] = wt[i-1] + bt[i-1]

for i in range(n):
    tat[i] = wt[i] + bt[i]

print("\nP\tBT\tWT\tTAT")

for i in range(n):
    print(p[i], "\t", bt[i], "\t", wt[i], "\t", tat[i])

print("Average WT =", sum(wt)/n)
print("Average TAT =", sum(tat)/n)

. Difference between FCFS and SJF
FCFS	SJF
First process gets CPU first	Shortest process gets CPU first
Uses arrival order	Uses burst time
Simple to implement	Requires sorting
May have higher waiting time	Usually lower average waiting time
Can cause convoy effect	Long processes may suffer starvation
