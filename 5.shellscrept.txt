Multiplication table using FOR loop
#!/bin/bash

echo "Enter a number:"
read n

for ((i=1; i<=10; i++))
do
    echo "$n x $i = $((n*i))"
done

#!/bin/bash

echo "Enter a number:"
read n

i=1

until [ $i -gt 10 ]
do
    echo "$n x $i = $((n*i))"
    i=$((i+1))
done

Check positive, negative or zero

echo "Enter a number:"
read n

if [ $n -gt 0 ]
then
    echo "Positive number"

elif [ $n -lt 0 ]
then
    echo "Negative number"

else
    echo "Zero"

fi



heck whether a file exists and display information

#!/bin/bash

echo "Enter filename:"
read file

if [ -f "$file" ]
then
    echo "File exists"
    ls -lh "$file"
else
    echo "File does not exist"
fi
